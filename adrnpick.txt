{$CLEO .cs}
{$NOSOURCE}
{$USE CLEO+}
script_name 'adrnpk'

int count
float x,y,z
int vMakepicks = False
int current_pick
int pickup_mem = allocate_memory 36 // 9 pickup vars * 4 bytes (int)

count = generate_random_int_in_range 4500 5500
wait count // idk how long until the pickups are made

set_script_event_save_confirmation True @SaveConfirm count
// last param needs to be a var or something, idk, wouldn't work otherwise

:makepicks
count = 0
while count < 9
switch count
// LS
case 0 // E6 : Jefferson : On roof of hospital
x = 2055.0
y = -1389.0
z = 24.0
case 1 // F4 : Santa Maria Beach : At foot of pier's ferris wheel
x = 364.0
y = -2055.0
z = 15.0
// SF
case 2 // C1 : City Hall : In doorway
x = -2802.0
y = 375.0
z = 6.0
case 3 // D2 : Foster Valley : Outside arena
x = -2023.0
y = -406.0
z = 39.0
// LV
case 4 // A6 : KACC : In corner of fence behind packing crates
x = 2503.0
y = 2823.0
z = 11.0
case 5 // B6 : The Ring Master : In corner of building
x = 2300.0
y = 1820.0
z = 16.0
case 6 // B6 : Pilgrim : Middle of alleyway
x = 2468.0
y = 1787.0
z = 11.0
case 7 // B5 : The High Roller : In corner on roof
x = 1947.0
y = 1333.0
z = 16.0
case 8 // B5 : Last Dime Motel : In corner, slightly hidden in flowers
x = 1955.0
y = 665.0
z = 14.0
end

current_pick = create_pickup #adrenaline PickupType.OnStreetSlow x y z
write_struct_param pickup_mem count current_pick
// make the pickup and write the handle to memory, offset by the current `count` number
// there's 9 pickups, need to keep track of them somehow

count++
end // while

vMakepicks = False

:pick_loop // after it's done making the picks, it'll wait in case it has to make them again
wait 5000
if vMakepicks == True
then
gosub @makepicks
vMakepicks = False
end
goto @pick_loop

:SaveConfirm // goes here when the savegame menu pops up
count = 0
while count < 9
current_pick = read_struct_param pickup_mem count
remove_pickup current_pick // read the handles we wrote to memory and remove the pickups
count++
end
vMakepicks = True
return_script_event
