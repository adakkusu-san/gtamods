{$CLEO .cs}
{$NOSOURCE}
{$USE CLEO+}
script_name 'adrnpk'

int iTemp
float x,y,z
int current_pick
int pickup_mem = get_label_pointer @pickmem

set_script_event_save_confirmation True @SaveConfirm current_pick
// last param needs to be a var or something, idk, wouldn't work otherwise

iTemp = get_current_save_slot
if iTemp > 0 // wait until the default pickups are done
then wait 10
else wait 3500
end

:makepicks
iTemp = 0
while iTemp < 9
switch_start iTemp count 8 has_def True go @c_0 incase 1 go @c_1 incase 2 go @c_2 incase 3 go @c_3 incase 4 go @c_4 incase 5 go @c_5 incase 6 go @c_6 incase 7 go @c_7
switch_continued incase 8 go @c_8 skip 99 9 99 9 99 9 99 9 99 9 99 9 99 9 99 9

// LS
:c_0 // E6 : Jefferson : On roof of hospital
x = 2055.0
y = -1389.0
z = 24.0
goto @cEnd
:c_1 // F4 : Santa Maria Beach : At foot of pier's ferris wheel
x = 364.0
y = -2055.0
z = 15.0
goto @cEnd
// SF
:c_2 // C1 : City Hall : In doorway
x = -2802.0
y = 375.0
z = 6.0
goto @cEnd
:c_3 // D2 : Foster Valley : Outside arena
x = -2023.0
y = -406.0
z = 39.0
goto @cEnd
// LV
:c_4 // A6 : KACC : In corner of fence behind packing crates
x = 2503.0
y = 2823.0
z = 11.0
goto @cEnd
:c_5 // B6 : The Ring Master : In corner of building
x = 2300.0
y = 1820.0
z = 16.0
goto @cEnd
:c_6 // B6 : Pilgrim : Middle of alleyway
x = 2468.0
y = 1787.0
z = 11.0
goto @cEnd
:c_7 // B5 : The High Roller : In corner on roof
x = 1947.0
y = 1333.0
z = 16.0
goto @cEnd
:c_8 // B5 : Last Dime Motel : In corner, slightly hidden in flowers
x = 1955.0
y = 665.0
z = 14.0
:cEnd

current_pick = create_pickup #adrenaline PickupType.OnStreetSlow x y z
write_struct_param pickup_mem iTemp current_pick
// make the pickup and write the handle to memory, offset by the current `iTemp` number
// there's 9 pickups, need to keep track of them somehow

iTemp++
end // while

:pick_loop // after it's done making the picks, it'll wait in case it has to make them again
wait 1234
if iTemp <= 0
then
goto @makepicks
end
goto @pick_loop

:SaveConfirm // goes here when the savegame menu pops up
// iTemp is currently the total amount
while iTemp > 0
iTemp--
current_pick = read_struct_param pickup_mem iTemp
remove_pickup current_pick // read the handles we wrote to memory and remove the pickups
end
return_script_event

:pickmem
hex
00(36) // 9 pickup vars * 4 bytes (int)
end
