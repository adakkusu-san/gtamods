{$CLEO .cs}
{$NOSOURCE}
script_name 'coord'
wait 2000

int c,d,e,a,p
float b,x,y,z,h
string g

add_text_label 'MSZP_A' "~1~.~1~"
add_text_label 'MSZP_0' "~1~.0~1~"
add_text_label 'MSZP_N' "-~1~.~1~"
add_text_label 'MSZPN0' "-~1~.0~1~"

while get_fading_status
wait 0
end

:coord_loop
if not is_player_playing $scplayer
then wait 1000
goto @coord_loop
end

if not is_char_in_any_car $scplayer
then
get_char_coordinates $scplayer x y z
get_char_heading $scplayer h
else
store_car_char_is_in_no_save $scplayer p
get_car_coordinates p x y z
get_car_heading p h
end

get_area_visible a

TIMERA = 0
TIMERB = 0
while TIMERA < 500
wait 0
use_text_commands True

cleo_call @float2int 1 x d e c
b = 210.0
gosub @text_format
display_text_with_2_numbers 610.0 b g d e

cleo_call @float2int 1 y d e c
b += 10.0
gosub @text_format
display_text_with_2_numbers 610.0 b g d e

cleo_call @float2int 1 z d e c
b += 10.0
gosub @text_format
display_text_with_2_numbers 610.0 b g d e

cleo_call @float2int 1 h d e c
b += 10.0
gosub @text_format
display_text_with_2_numbers 610.0 b g d e

b += 20.0
gosub @text_format
display_text_with_number 610.0 b 'NUMBER' a

// all that for this
// display_text_formatted 610.0 210.0 "%.2f~n~%.2f~n~%.2f~n~%.2f~n~%d" x y z h a  

use_text_commands False
end

goto @coord_loop

:text_format
switch c
case 0
g = 'MSZP_A'
case 1
g = 'MSZP_0'
case 2
g = 'MSZP_N'
case 3
g = 'MSZPN0'
end
set_text_wrapx 640.0
set_text_right_justify True
set_text_draw_before_fade False
return

:float2int
// original from: https://gtamods.com/wiki/Display_float_in_text
0092: 1@ = float 0@ to_integer
0093: 3@ = integer 1@ to_float
0087: 2@ = 0@ 
0063: 2@ -= 3@
0013: 2@ *= 100.0 
0092: 4@ = float 2@ to_integer
0095: make 4@ absolute_integer
if and
0.0 > 0@
0@ > -1.0
then
if
4@ > 9
then
5@ = 2
else
5@ = 3
end
else
if
4@ > 9
then
5@ = 0
else
5@ = 1
end
end
cleo_return 3 1@ 4@ 5@
