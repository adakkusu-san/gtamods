{$CLEO .cs}
{$NOSOURCE}
{$USE ini}
script_name 'mszclth'

wait 2000

int count
longstring current_part,model_name,texture_name

if not does_file_exist "cleo\mszcloth.ini"
then
print_help_string "mszcloth.ini~n~not found"
goto @end_of_script
end

while not is_player_playing $player1
wait 0
end

count = 0
while count < 17
if count == 4 // not implemented yet
then count = 13
end
string_format current_part "%d" count
model_name = read_string_from_ini_file "cleo\mszcloth.ini" current_part "dff"
if not model_name == "skip"
then
if model_name == "remove"
then
give_player_clothes_outside_shop $player1 0 0 count
else
texture_name = read_string_from_ini_file "cleo\mszcloth.ini" current_part "txd"
give_player_clothes_outside_shop $player1 texture_name model_name count
end
end
count++
end

build_player_model $player1

:end_of_script
terminate_this_custom_script
