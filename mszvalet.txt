:start_of_script

while true
wait 250
if is_key_pressed 0xc0
then
break
end
end

if not is_player_playing $scplayer
then
goto @start_of_script
end

0@ = create_menu 'VALET' 30.0 140.0 200.0 1 True True Align.Left
set_menu_column_orientation 0@ 0 Align.Left
set_menu_column 0@ 0 'FED_BL1' 'LA' 'SF' 'VE' 'VALET' 'LA' 'SF' 'VE' 'FEC_VES' 'KART' 'DUMMY' 'DUMMY' 'DUMMY'

wait 250

if is_button_pressed 0 Button.Cross // Select
then

switch 1@
case 0
gosub @LS_T
case 1
gosub @SF_T
case 2
gosub @LV_T

case 4
gosub @LS_V
case 5
gosub @SF_V
case 6
gosub @LV_V

case 8
gosub @kart_spawn
end

break

end

then
break
end

end

delete_menu 0@

set_player_control $player1 True
set_everyone_ignore_player $player1 False

goto @start_of_script

:check_player
while not is_player_playing $scplayer
wait 250
end
