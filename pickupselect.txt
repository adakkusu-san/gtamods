{$CLEO .cs}
{$NOSOURCE}
{$USE INI,CLEO+}
script_name 'MSZPIKA'
wait 2000

int iTemp,count

float pickXf,pickYf,pickZf
int pickXi,pickYi,pickZi

int pickpoint,keypoint

// TODO: use extravars for some of these
int gvar,a_index_max

int v_index,a_index,pickarray,bliptoadd

int which_key_bit,bStatus // bitwise
// ran out of variables to use, had to do bitwise
// after I figure out how, I'll make those variables into a single one

longstring picktype
longstring str

:start_of_script
if not does_file_exist "cleo\pickupselect.ini"
then
print_string_now "pickupselect.ini not found" 4000
goto @end_of_script
end
wait 2000
TIMERA = 0
TIMERB = 0
while TIMERA < 2000
wait 0
use_text_commands True
display_text 490.0 10.0 'MSZPIKA'
use_text_commands False
end

add_text_label 'MSZPIKA' "pickup select"
add_text_label 'MSZPS01' "step = 1.0"
add_text_label 'MSZPS10' "step = 10.0"
add_text_label 'MSZPST' "speed = TRUE"
add_text_label 'MSZPSF' "speed = FALSE"
add_text_label 'MSZARRN' "number = ~1~"
add_text_label 'MSZ_X' "X ="
add_text_label 'MSZ_Y' "Y ="
add_text_label 'MSZ_Z' "Z ="
add_text_label 'MSZSAVE' "saved to INI"

const
bInvalid = 0
bGetCoords = 1
bWriteCoords = 2
bWritten = 3
bStep = 4
bSpeed = 5
end

const
pUpKey = 0
pLeftKey = 1
pDownKey = 2
pRightKey = 3
pAboveKey = 4
pBelowKey = 5
pStepKey = 6
pSpeedKey = 7
pReloadKey = 8
pGroundZKey = 9
pSaveKey = 10
pTeleportKey = 11
end

keypoint = get_label_pointer @key_vars
count = 0
while count < 12
switch_start count 12 has_def False @k_end incase pUpKey go @k_Up incase pLeftKey go @k_Left incase pDownKey go @k_Down incase pRightKey go @k_Right incase pAboveKey go @k_Above incase pBelowKey go @k_Below incase pStepKey go @k_Step
switch_continued incase pSpeedKey go @k_Speed incase pReloadKey go @k_Reload incase pGroundZKey go @k_GroundZ incase pSaveKey go @k_Save incase pTeleportKey go @k_Teleport skip -1 @k_end -1 @k_end -1 @k_end -1 @k_end
:k_Up
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "up"
then iTemp = 0x49 // I
end
goto @k_end
:k_Left
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "left"
then iTemp = 0x4a // J
end
goto @k_end
:k_Down
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "down"
then iTemp = 0x4b // K
end
goto @k_end
:k_Right
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "right"
then iTemp = 0x4c // L
end
goto @k_end
:k_Above
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "above"
then iTemp = 0x55 // U
end
goto @k_end
:k_Below
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "below"
then iTemp = 0x4f // O
end
goto @k_end
:k_Step
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "step"
then iTemp = 0x4d // M
end
goto @k_end
:k_Speed
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "speed"
then iTemp = 0xbe // .
end
goto @k_end
:k_Reload
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "reload"
then iTemp = 0x58 // X
end
goto @k_end
:k_GroundZ
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "groundz"
then iTemp = 0x42 // B
end
goto @k_end
:k_Save
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "save"
then iTemp = 0xbc // ,
end
goto @k_end
:k_Teleport
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "teleport"
then iTemp = 0x54 // T
end
:k_end
write_struct_offset keypoint count 1 iTemp
count++
end

picktype = read_string_from_ini_file "cleo\pickupselect.ini" "config" "picktype"
switch picktype
case "weapon"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "weapon" "offset"
then gvar = 3032
end
if not a_index_max = read_int_from_ini_file "cleo\pickupselect.ini" "weapon" "maxval"
then a_index_max = 217
end
case "flower"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "flower" "offset"
then gvar = 3257
end
if not a_index_max = read_int_from_ini_file "cleo\pickupselect.ini" "flower" "maxval"
then a_index_max = 40
end
case "armour"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "armour" "offset"
then gvar = 3297
end
if not a_index_max = read_int_from_ini_file "cleo\pickupselect.ini" "armour" "maxval"
then a_index_max = 48
end
case "bribe"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "bribe" "offset"
then gvar = 3345
end
if not a_index_max = read_int_from_ini_file "cleo\pickupselect.ini" "bribe" "maxval"
then a_index_max = 50
end
default
gvar = read_int_from_ini_file "cleo\pickupselect.ini" picktype "offset"
a_index_max = read_int_from_ini_file "cleo\pickupselect.ini" picktype "maxval"
end

if not v_index = read_int_from_ini_file "cleo\pickupselect.ini" picktype "index"
then v_index = 0
end

a_index = v_index
v_index += gvar
pickarray = &0(v_index,16384i)

set_local_var_bit_const bStatus bInvalid
set_local_var_bit_const bStatus bGetCoords
clear_local_var_bit_lvar bStatus bWriteCoords
clear_local_var_bit_lvar bStatus bWritten
clear_local_var_bit_lvar bStatus bStep
set_local_var_bit_const bStatus bSpeed

gosub @pick_check

:pick_loop
wait 0
use_text_commands True

gosub @text_display

// gosub @select_pick // TODO 

if is_local_var_bit_set_const bStatus bInvalid
then goto @skip_it_all
end

gosub @which_key_press

if is_local_var_bit_set_const bStatus bWriteCoords
then gosub @write_coords
clear_local_var_bit_lvar bStatus bWriteCoords
end

iTemp = read_struct_offset keypoint pSaveKey 1
if is_key_pressed iTemp
then
gosub @save_coords
end

:skip_it_all
use_text_commands False
goto @pick_loop

:text_display
gosub @text_setup

if not is_local_var_bit_set_const bStatus bStep
then
display_text 490.0 30.0 'MSZPS01'
else
display_text 490.0 30.0 'MSZPS10'
end

gosub @text_setup

if is_local_var_bit_set_const bStatus bSpeed
then
display_text 490.0 50.0 'MSZPST'
else
display_text 490.0 50.0 'MSZPSF'
end

gosub @text_setup
display_text_with_number 490.0 70.0 'MSZARRN' a_index

get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
pickXi =# pickXf
pickYi =# pickYf
pickZi =# pickZf

gosub @text_setup
display_text 440.0 100.0 'MSZ_X'
gosub @text_setup
set_text_proportional False
display_text_with_number 490.0 100.0 'NUMBER' pickXi

gosub @text_setup
display_text 440.0 110.0 'MSZ_Y'
gosub @text_setup
set_text_proportional False
display_text_with_number 490.0 110.0 'NUMBER' pickYi

gosub @text_setup
display_text 440.0 120.0 'MSZ_Z'
gosub @text_setup
set_text_proportional False
display_text_with_number 490.0 120.0 'NUMBER' pickZi

return

:text_setup
set_text_wrapx 640.0
set_text_right_justify True
return

:select_pick
/*
count = 0
if or
is_button_pressed 0 Button.DpadUp // G
is_button_pressed 0 Button.DpadDown // H
is_button_pressed 0 Button.DpadLeft // N
is_button_pressed 0 Button.DpadRight // Y
then
count = 1
end

if is_button_pressed 0 Button.DpadUp // G
then
a_index--
if a_index < 0
then
a_index = a_index_max
a_index--
end
end
if is_button_pressed 0 Button.DpadDown // H
then
a_index++
if a_index >= a_index_max
then
a_index = 0
end
end
if is_button_pressed 0 Button.DpadLeft // N
then
a_index -= 10
if a_index < 0
then
a_index += a_index_max
end
end
if is_button_pressed 0 Button.DpadRight // Y
then
a_index += 10
if a_index >= a_index_max
then
a_index -= a_index_max
end
end

if count > 0
then
v_index = a_index
v_index += gvar
pickarray = &0(v_index,16384i)
end

while count > 0
wait 100
if and
not is_button_pressed 0 Button.DpadUp // G
not is_button_pressed 0 Button.DpadDown // H
not is_button_pressed 0 Button.DpadLeft // N
not is_button_pressed 0 Button.DpadRight // Y
then
count = 0
end
end
*/
return

:pick_check
remove_blip bliptoadd
if does_pickup_exist pickarray
then
bliptoadd = add_blip_for_pickup pickarray
clear_local_var_bit_lvar bStatus bInvalid
else
set_local_var_bit_const bStatus bInvalid
end
return

:get_coords
pickpoint = get_pickup_pointer pickarray
pickXi = read_struct_offset pickpoint 0x10 2
pickYi = read_struct_offset pickpoint 0x12 2
pickZi = read_struct_offset pickpoint 0x14 2
return

:which_key_press
iTemp = read_struct_offset keypoint pReloadKey 1
if is_key_pressed iTemp
then
print_string_now "reloaded pickup select" 2000
gosub @wait_key_release
goto @start_of_script
end

which_key_bit = 0
clear_local_var_bit_const bStatus bWriteCoords
count = 0
while count < 6
iTemp = read_struct_offset keypoint count 1
if is_key_pressed iTemp
then
set_local_var_bit_lvar which_key_bit count
set_local_var_bit_const bStatus bGetCoords
set_local_var_bit_const bStatus bWriteCoords
break
end
count++
end

if which_key_bit > 0
then
if not is_local_var_bit_set_const bStatus bSpeed
then
if is_local_var_bit_set_const bStatus bWritten
then
clear_local_var_bit_const bStatus bWriteCoords
which_key_bit = 0
end
end
else
clear_local_var_bit_const bStatus bWritten
end

if is_local_var_bit_set_const bStatus bGetCoords
then gosub @get_coords
clear_local_var_bit_const bStatus bGetCoords
end

iTemp = read_struct_offset keypoint pStepKey 1
if is_key_pressed iTemp
then
if not is_local_var_bit_set_const bStatus bStep
then set_local_var_bit_lvar bStatus bStep
else clear_local_var_bit_const bStatus bStep
end
gosub @wait_key_release
end

iTemp = read_struct_offset keypoint pSpeedKey 1
if is_key_pressed iTemp
then
if is_local_var_bit_set_const bStatus bSpeed
then clear_local_var_bit_lvar bStatus bSpeed
else set_local_var_bit_const bStatus bSpeed
end
gosub @wait_key_release
end

if which_key_bit > 0
then
count = 0
while count < 6
if is_local_var_bit_set_lvar which_key_bit count
then
if not is_local_var_bit_set_const bStatus bStep
then iTemp = 8
else iTemp = 80
end
switch_start count 6 has_def False @c_end incase 0 go @c_0 incase 1 go @c_1 incase 2 go @c_2 incase 3 go @c_3 incase 4 go @c_4 incase 5 go @c_5 skip -1 @c_end
:c_0 // up
pickYi += iTemp
goto @c_end

:c_1 // left
pickXi -= iTemp
goto @c_end

:c_2 // down
pickYi -= iTemp
goto @c_end

:c_3 // right
pickXi += iTemp
goto @c_end

:c_4 // above
pickZi += iTemp
goto @c_end

:c_5 // below
pickZi -= iTemp
:c_end

end//if
count++
end//while
end//if

iTemp = read_struct_offset keypoint pGroundZKey 1
if is_key_pressed iTemp
then
which_key_bit = 0
set_local_var_bit_lvar which_key_bit pGroundZKey
set_local_var_bit_const bStatus bWriteCoords
gosub @wait_key_release
end

iTemp = read_struct_offset keypoint pTeleportKey 1
if is_key_pressed iTemp
then
do_fade 500 Fade.Out
pickpoint = get_pickup_pointer pickarray
get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
get_pickup_coordinates pickarray store_to pickXi pickYi pickZi
while get_fading_status
wait 250
end
pickZf += 10.0
load_scene pickXf pickYf pickZf
request_collision pickXf pickYf
load_all_models_now // hotfix // remove later
get_closest_char_node pickXi pickYi pickZi store_to pickXf pickYf pickZf
set_char_coordinates $scplayer pickXf pickYf pickZf
do_fade 500 Fade.In
gosub @wait_key_release
end

return

:write_coords
if which_key_bit > 0
then
pickpoint = get_pickup_pointer pickarray
if not is_local_var_bit_set_lvar which_key_bit pGroundZKey
then
write_struct_offset pickpoint 0x10 2 pickXi
write_struct_offset pickpoint 0x12 2 pickYi
write_struct_offset pickpoint 0x14 2 pickZi
else
get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
pickZf = get_ground_z_for_3d_coord pickXf pickYf pickZf
pickZf += 2.0
pickZi =# pickZf
pickZi *= 8

write_struct_offset pickpoint 0x14 2 pickZi

iTemp = read_struct_offset keypoint pGroundZKey 1
gosub @wait_key_release
end

set_local_var_bit_const bStatus bWritten
end
return

:save_coords
get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
string_format str "%d" pickarray
write_int_to_ini_file v_index "cleo\pickupselect.ini" str "global_variable"
write_int_to_ini_file a_index "cleo\pickupselect.ini" str "a_index"
write_float_to_ini_file pickXf "cleo\pickupselect.ini" str "x"
write_float_to_ini_file pickYf "cleo\pickupselect.ini" str "y"
write_float_to_ini_file pickZf "cleo\pickupselect.ini" str "z"

TIMERA = 0
TIMERB = 0
while TIMERA < 1000
wait 0
display_text_with_number 490.0 200.0 'MSZSAVE' a_index
end

gosub @wait_key_release
return

:wait_key_release
while is_key_pressed iTemp
wait 0
end
return

:end_of_script
terminate_this_custom_script

:key_vars
hex
00 00 00 00 00 00 00 00 00 00 00 00
//00 00 00 00
end

4=above
5=below
6=step
7=speed
8=reload
9=ground
10=save
11=teleport
12=prev
13=next
14=prev-10
15=next+10
*/
