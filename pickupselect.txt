{$CLEO .cs}
{$NOSOURCE}
{$USE INI,CLEO+}
script_name 'MSZPIKA'
wait 2000

int iTemp
float fTemp
int count

float pickXf,pickYf,pickZf
int pickXi,pickYi,pickZi

int pickpoint,keypoint

// TODO: use extravars for some of these
int gvar,maxval

int index,index_array,pickarray,bliptoadd

int step,speed // TODO use bitwise
int which_key_bit
int bStatus // bitwise / 0=invalid, 1=getcoords, 2=writecoords, 3=wrote
// ran out of variables to use, had to do bitwise
// after I figure out how, I'll make those variables into a single one

longstring picktype
longstring str

add_text_label 'MSZPIKA' "pickup select"
add_text_label 'MSZPS01' "step = 1.0"
add_text_label 'MSZPS10' "step = 10.0"
add_text_label 'MSZPST' "speed = TRUE"
add_text_label 'MSZPSF' "speed = FALSE"
add_text_label 'MSZARRN' "number = ~1~"
add_text_label 'MSZ_X' "X ="
add_text_label 'MSZ_Y' "Y ="
add_text_label 'MSZ_Z' "Z ="
add_text_label 'MSZSAVE' "saved to INI"

step = 8
speed = True

:start_of_script
if not does_file_exist "cleo\pickupselect.ini"
then
print_string_now "pickupselect.ini not found" 4000
goto @end_of_script
end
wait 2000
TIMERA = 0
TIMERB = 0
while TIMERA < 2000
wait 0
use_text_commands True
display_text 490.0 10.0 'MSZPIKA'
use_text_commands False
end

keypoint = get_label_pointer @key_vars
count = 0
while count < 12
switch count // TODO: do it with the command thing instead of this
case 0
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "up"
then iTemp = 0x49 // I
end
case 1
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "left"
then iTemp = 0x4a // J
end
case 2
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "down"
then iTemp = 0x4b // K
end
case 3
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "right"
then iTemp = 0x4c // L
end
case 4
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "above"
then iTemp = 0x55 // U
end
case 5
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "below"
then iTemp = 0x4f // O
end
case 6
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "step"
then iTemp = 0x4d // M
end
case 7
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "speed"
then iTemp = 0xbe // .
end
case 8
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "reload"
then iTemp = 0x58 // X
end
case 9
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "groundz"
then iTemp = 0x42 // B
end
case 10
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "save"
then iTemp = 0xbc // ,
end
case 11
if not iTemp = read_int_from_ini_file "cleo\pickupselect.ini" "config" "teleport"
then iTemp = 0x54 // T
end
end
write_struct_offset keypoint count 1 iTemp
count++
end

picktype = read_string_from_ini_file "cleo\pickupselect.ini" "config" "picktype"
switch picktype
case "weapon"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "weapon" "offset"
then gvar = 3032
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" "weapon" "maxval"
then maxval = 217
end
case "flower"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "flower" "offset"
then gvar = 3257
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" "flower" "maxval"
then maxval = 40
end
case "armour"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "armour" "offset"
then gvar = 3297
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" "armour" "maxval"
then maxval = 48
end
case "bribe"
if not gvar = read_int_from_ini_file "cleo\pickupselect.ini" "bribe" "offset"
then gvar = 3345
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" "bribe" "maxval"
then maxval = 50
end
default
gvar = read_int_from_ini_file "cleo\pickupselect.ini" picktype "offset"
maxval = read_int_from_ini_file "cleo\pickupselect.ini" picktype "maxval"
end

if not index = read_int_from_ini_file "cleo\pickupselect.ini" picktype "index"
then index = 0
end

index_array = index
index += gvar
pickarray = &0(gvar,16384i)

set_local_var_bit_const bStatus 0 // invalid
set_local_var_bit_const bStatus 1 // getcoords
clear_local_var_bit_lvar bStatus 2 // writecoords
clear_local_var_bit_lvar bStatus 3 // wrotethem

gosub @pick_check

:pick_loop
wait 0
use_text_commands True

gosub @text_display

// gosub @select_pick // TODO

if is_local_var_bit_set_const bStatus 0 // invalid
then goto @skip_it_all
end

gosub @which_key_press

if is_local_var_bit_set_const bStatus 2
then gosub @write_coords
clear_local_var_bit_lvar bStatus 2
end

iTemp = read_struct_offset keypoint 10 1
if is_key_pressed iTemp
then
gosub @save_coords
end

:skip_it_all
use_text_commands False
goto @pick_loop

:text_display
gosub @text_setup
if step == 8
then
display_text 490.0 30.0 'MSZPS01'
else
display_text 490.0 30.0 'MSZPS10'
end

gosub @text_setup
if speed == True
then
display_text 490.0 50.0 'MSZPST'
else
display_text 490.0 50.0 'MSZPSF'
end

gosub @text_setup
display_text_with_number 490.0 70.0 'MSZARRN' index_array

get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
pickXi =# pickXf
pickYi =# pickYf
pickZi =# pickZf

gosub @text_setup
display_text 440.0 100.0 'MSZ_X'
gosub @text_setup
set_text_proportional False
display_text_with_number 490.0 100.0 'NUMBER' pickXi

gosub @text_setup
display_text 440.0 110.0 'MSZ_Y'
gosub @text_setup
set_text_proportional False
display_text_with_number 490.0 110.0 'NUMBER' pickYi

gosub @text_setup
display_text 440.0 120.0 'MSZ_Z'
gosub @text_setup
set_text_proportional False
display_text_with_number 490.0 120.0 'NUMBER' pickZi

return

:text_setup
set_text_wrapx 640.0
set_text_right_justify True
return

:select_pick
// not done yet
return

:pick_check
remove_blip bliptoadd
if does_pickup_exist pickarray
then
bliptoadd = add_blip_for_pickup pickarray
clear_local_var_bit_lvar bStatus 0
else
set_local_var_bit_const bStatus 0 // invalid
end
return

:get_coords
pickpoint = get_pickup_pointer pickarray
pickXi = read_struct_offset pickpoint 0x10 2
pickYi = read_struct_offset pickpoint 0x12 2
pickZi = read_struct_offset pickpoint 0x14 2
return

:which_key_press
iTemp = read_struct_offset keypoint 8 1
if is_key_pressed iTemp
then
print_string_now "reloaded pickup select" 2000
gosub @wait_key_release
goto @start_of_script
end

which_key_bit = 0
clear_local_var_bit_const bStatus 2 // write
count = 0
while count < 6
iTemp = read_struct_offset keypoint count 1
if is_key_pressed iTemp
then
set_local_var_bit_lvar which_key_bit count
set_local_var_bit_const bStatus 1 // read
set_local_var_bit_const bStatus 2 // write
break
end
count++
end

if which_key_bit > 0
then
if speed == False
then
if is_local_var_bit_set_const bStatus 3
then
clear_local_var_bit_const bStatus 2 // write
which_key_bit = 0
end
end
else
clear_local_var_bit_const bStatus 3 // wrote
end

if is_local_var_bit_set_const bStatus 1
then gosub @get_coords
clear_local_var_bit_const bStatus 1
end

iTemp = read_struct_offset keypoint 6 1
if is_key_pressed iTemp
then
if step == 8
then step = 80
else step = 8
end
gosub @wait_key_release
end

iTemp = read_struct_offset keypoint 7 1
if is_key_pressed iTemp
then
if speed == True
then speed = False
else speed = True
end
gosub @wait_key_release
end

if which_key_bit > 0
then
count = 0
while count < 6
if is_local_var_bit_set_lvar which_key_bit count
then
switch count // TODO: do it with the command thing instead of this
case 0 // up
pickYi += step
case 1 // left
pickXi -= step
case 2 // down
pickYi -= step
case 3 // right
pickXi += step
case 4 // above
pickZi += step
case 5 // below
pickZi -= step
end//switch
end//if
count++
end//while
end//if

iTemp = read_struct_offset keypoint 9 1 // ground_z
if is_key_pressed iTemp
then
which_key_bit = 0
set_local_var_bit_lvar which_key_bit 6
set_local_var_bit_const bStatus 2
gosub @wait_key_release
end

iTemp = read_struct_offset keypoint 11 1 // teleport
if is_key_pressed iTemp
then
do_fade 500 Fade.Out
pickpoint = get_pickup_pointer pickarray
get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
get_pickup_coordinates pickarray store_to pickXi pickYi pickZi
while get_fading_status
wait 250
end
load_scene pickXf pickYf pickZf
request_collision pickXf pickYf
get_closest_char_node pickXi pickYi pickZi store_to pickXf pickYf pickZf
pickZf += 2.0
fTemp = get_ground_z_for_3d_coord pickXf pickYf pickZf
set_char_coordinates $scplayer pickXf pickYf fTemp
do_fade 500 Fade.In
gosub @wait_key_release
end

return

:write_coords
if which_key_bit > 0
then
pickpoint = get_pickup_pointer pickarray
if not is_local_var_bit_set_lvar which_key_bit 6 // ground_z
then
write_struct_offset pickpoint 0x10 2 pickXi
write_struct_offset pickpoint 0x12 2 pickYi
write_struct_offset pickpoint 0x14 2 pickZi
else
get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
fTemp = get_ground_z_for_3d_coord pickXf pickYf pickZf
fTemp += 2.0
pickZi =# fTemp
pickZi *= 8

write_struct_offset pickpoint 0x14 2 pickZi

iTemp = read_struct_offset keypoint 9 1 // ground_z
gosub @wait_key_release
end

set_local_var_bit_const bStatus 3 // wrotethem
end
return

:save_coords
get_pickup_coordinates pickarray store_to pickXf pickYf pickZf
string_format str "%d" pickarray
write_int_to_ini_file index "cleo\pickupselect.ini" str "global_variable"
write_int_to_ini_file index_array "cleo\pickupselect.ini" str "index_array"
write_float_to_ini_file pickXf "cleo\pickupselect.ini" str "x"
write_float_to_ini_file pickYf "cleo\pickupselect.ini" str "y"
write_float_to_ini_file pickZf "cleo\pickupselect.ini" str "z"

TIMERA = 0
TIMERB = 0
while TIMERA < 1000
wait 0
display_text_with_number 490.0 200.0 'MSZSAVE' index_array
end

gosub @wait_key_release
return

:wait_key_release
while is_key_pressed iTemp
wait 0
end
return

:end_of_script
terminate_this_custom_script

:key_vars
hex
00 00 00 00 00 00 00 00 00 00 00 00
end
/*
iTemp = read_struct_offset keypoint v 1
0=up
1=left
2=down
3=right
4=above
5=below
6=step
7=speed
8=reload
9=ground
10=save
11=teleport
*/
