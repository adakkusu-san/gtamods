{$CLEO .cs}
{$NOSOURCE}
{$USE INI,CLEO+}
script_name 'MSZPIKA'
wait 2000

int index
int indexnum
int pickarray

float pickX
float pickY
float pickZ

int bliptoadd
int pickhandle

int pickpoint

int readmem
int writemem
int pickXo
int pickYo
int pickZo
int pickAo

int maxval
int picktype
int step
int speed 
int which_key_pressed

float temp_float

string gxtname
longstring str

add_text_label 'MSZPIKA' "pickup select"

add_text_label 'MSZPS01' "step = 1.0"
add_text_label 'MSZPS10' "step = 10.0"

add_text_label 'MSZPST' "speed = TRUE"
add_text_label 'MSZPSF' "speed = FALSE"

add_text_label 'MSZARRN' "number = ~1~"

add_text_label 'MSZ_X' "X ="
add_text_label 'MSZ_Y' "Y ="
add_text_label 'MSZ_Z' "Z ="

add_text_label 'MSZSAVE' "saved to INI"


/*
add_text_label 'MSZP_A' "~1~.~1~"
add_text_label 'MSZP_0' "~1~.0~1~"
add_text_label 'MSZP_N' "-~1~.~1~"
add_text_label 'MSZPN0' "-~1~.0~1~"

add_text_label 'MSZIKP' "is key pressed"
add_text_label 'MSZNKP' "no key pressed"
*/

step = 8
speed = False

:start_of_script

if not does_file_exist "cleo\pickupselect.ini"
then
print_string_now "pickupselect.ini not found" 4000
goto @end_of_script
end

/*
while true
wait 100
if is_key_pressed 0x68 // numpad 8
then
break
end
end
*/

wait 2000
TIMERA = 0
while TIMERA < 2000
wait 0
use_text_commands True
display_text 490.0 10.0 'MSZPIKA'
use_text_commands False
end

/*
3032 = pickup_weapon
3257 = pickup_flowers
3297 = pickup_armour
3345 = pickup_bribe
*/


if not picktype = read_int_from_ini_file "cleo\pickupselect.ini" "config" "index"
then picktype = 0
end

switch picktype
case 0
str = "weapon"
if not readmem = read_int_from_ini_file "cleo\pickupselect.ini" str "offset"
then readmem = 3032
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" str "maxval"
then maxval = 217
end
case 1
str = "flower"
if not readmem = read_int_from_ini_file "cleo\pickupselect.ini" str "offset"
then readmem = 3257
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" str "maxval"
then maxval = 40
end
case 2
str = "armour"
if not readmem = read_int_from_ini_file "cleo\pickupselect.ini" str "offset"
then readmem = 3297
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" str "maxval"
then maxval = 48
end
case 3
str = "bribe"
if not readmem = read_int_from_ini_file "cleo\pickupselect.ini" str "offset"
then readmem = 3345
end
if not maxval = read_int_from_ini_file "cleo\pickupselect.ini" str "maxval"
then maxval = 50
end
end

if not index = read_int_from_ini_file "cleo\pickupselect.ini" str "index"
then index = 0
end

/*
indexnum = index

readmem *= 4
readmem += 0xA49960
// https://gtamods.com/wiki/Memory_Addresses_(SA)

index *= 4
readmem += index

pickarray = read_memory address readmem size 4 vp False
*/
indexnum = index
index += readmem
pickarray = &0(index,16384i)

if does_pickup_exist pickarray
then
remove_blip bliptoadd

bliptoadd = add_blip_for_pickup pickarray

while true
wait 0

use_text_commands True

gosub @text_display

gosub @get_coords

gosub @which_key_press

gosub @write_coords

gosub @save_coords

use_text_commands False

end

end
//end

//goto @start_of_script

:end_of_script
terminate_this_custom_script

:text_display
gosub @text_display_setup
if step == 8
then
display_text 490.0 30.0 'MSZPS01'
else
display_text 490.0 30.0 'MSZPS10'
end

gosub @text_display_setup
if speed == 1
then
display_text 490.0 50.0 'MSZPST'
else
display_text 490.0 50.0 'MSZPSF'
end

gosub @text_display_setup
display_text_with_number 490.0 70.0 'MSZARRN' indexnum


get_pickup_coordinates pickarray store_to pickX pickY pickZ
pickXo =# pickX
pickYo =# pickY
pickZo =# pickZ

gosub @text_display_setup
display_text 440.0 100.0 'MSZ_X'
gosub @text_display_setup
set_text_proportional False
display_text_with_number 490.0 100.0 'NUMBER' pickXo

gosub @text_display_setup
display_text 440.0 110.0 'MSZ_Y'
gosub @text_display_setup
set_text_proportional False
display_text_with_number 490.0 110.0 'NUMBER' pickYo

gosub @text_display_setup
display_text 440.0 120.0 'MSZ_Z'
gosub @text_display_setup
set_text_proportional False
display_text_with_number 490.0 120.0 'NUMBER' pickZo

return

:text_display_setup
set_text_wrapx 640.0
set_text_right_justify True
return

// maybe useful
// https://gtaforums.com/topic/989882-search-for-pickups-on-the-map/
// https://gtaforums.com/topic/409207-struct-of-car/
// https://gtamods.com/wiki/Saves_(GTA_SA)

:get_coords
pickpoint = get_pickup_pointer pickarray
pickpoint += 0x10
pickXo = read_memory address pickpoint size 2 vp False
pickpoint += 0x2
pickYo = read_memory address pickpoint size 2 vp False
pickpoint += 0x2
pickZo = read_memory address pickpoint size 2 vp False
pickpoint += 0x2
pickAo = read_memory address pickpoint size 4 vp False // align
return

:which_key_press
/*
i = 0x49 // up
j = 0x4a // left
k = 0x4b // down
l = 0x4c // right
u = 0x55 // above
o = 0x4f // below

I = Y++
J = X--
K = Y--
L = X++
U = Z++
O = Z--

speed = FALSE // stop after each
speed = TRUE // don't stop
*/

if is_key_pressed 0x58 // X
then
print_string_now "reloaded pickup select" 2000
while is_key_pressed 0x58
wait 0
end
goto @start_of_script
end

if and
not is_key_pressed 0x49 // I
not is_key_pressed 0x4a // J
not is_key_pressed 0x4b // K
not is_key_pressed 0x4c // L
not is_key_pressed 0x55 // U
not is_key_pressed 0x4f // O
then
which_key_pressed = 0
end

if which_key_pressed == 4
then
if speed == False
then
return
end
end

if is_key_pressed 0x4d // m
then
if step == 8
then step = 80
else step = 8
end
end

while is_key_pressed 0x4d // m
wait 0
end

if is_key_pressed 0xbe // .
then
if speed == True
then speed = False
else speed = True
end
end

while is_key_pressed 0xbe // .
wait 0
end

if is_key_pressed 0x49 // I
then
pickYo += step
which_key_pressed = 1 // up/down
end

if is_key_pressed 0x4a // J
then
pickXo -= step
which_key_pressed = 2 // left/right
end

if is_key_pressed 0x4b // K
then
pickYo -= step
which_key_pressed = 1 // up/down
end

if is_key_pressed 0x4c // L
then
pickXo += step
which_key_pressed = 2 // left/right
end

if is_key_pressed 0x55 // U
then
pickZo += step
which_key_pressed = 3 // above/below
end

if is_key_pressed 0x4f // O
then
pickZo -= step
which_key_pressed = 3 // above/below
end

if is_key_pressed 0x42 // B
then
which_key_pressed = 9 // ground Z
end

return

:write_coords
pickpoint = get_pickup_pointer pickarray

if which_key_pressed > 0
then

pickpoint += 0x10
write_memory address pickpoint size 4 value pickXo False
pickpoint += 0x2
write_memory address pickpoint size 4 value pickYo False
pickpoint += 0x2
write_memory address pickpoint size 4 value pickZo False

if which_key_pressed == 9 // ground z
then

//goto @skipthisfornow
pickpoint = get_pickup_pointer pickarray
pickpoint += 0x10
pickXo = read_memory address pickpoint size 2 vp False
pickpoint += 0x2
pickYo = read_memory address pickpoint size 2 vp False
pickpoint += 0x2
pickZo = read_memory address pickpoint size 2 vp False

pickX =# pickXo
pickY =# pickYo
pickZ =# pickZo

pickX /= 8.0
pickY /= 8.0
pickZ /= 8.0
//pickXo /= 8
//pickYo /= 8
//pickZo /= 8

temp_float = get_ground_z_for_3d_coord pickX pickY pickZ
temp_float += 2.0
pickZo =# temp_float
pickZo *= 8

write_memory address pickpoint size 4 value pickZo vp False


//print_formatted "%f.0 %f.0 %f.0 %d %d %d" 2000 pickX pickY pickZ pickXo pickYo pickZo temp_float
/*
:skipthisfornow

get_char_coordinates $scplayer store_to pickX pickY pickZ
temp_float = get_ground_z_for_3d_coord pickX pickY pickZ
print_formatted "%1.f %1.f %1.f~n~%1.f" 2000 pickX pickY pickZ temp_float 
*/

while is_key_pressed 0x42
wait 0
end

end

pickpoint += 0x2
write_memory address pickpoint size 4 value pickAo False // align

which_key_pressed = 4
end

return

:save_coords

if is_key_pressed 0xbc // ,
then
pickhandle = get_pickup_this_coord pickX pickY pickZ True

get_pickup_coordinates pickarray store_to pickX pickY pickZ


string_format str "%d" picktype
write_int_to_ini_file index "cleo\pickupselect.ini" str "global_variable"
write_int_to_ini_file indexnum "cleo\pickupselect.ini" str "array_index"

write_float_to_ini_file pickX "cleo\pickupselect.ini" str "x"
write_float_to_ini_file pickY "cleo\pickupselect.ini" str "y"
write_float_to_ini_file pickZ "cleo\pickupselect.ini" str "z"

TIMERA = 0
TIMERB = 0
while TIMERA < 1000
wait 0
display_text_with_number 490.0 200.0 'MSZSAVE' indexnum
end


while is_key_pressed 0xbc
wait 0
end

end

return
