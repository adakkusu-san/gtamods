{$CLEO .cs}
{$NOSOURCE}
{$USE ini,CLEO+}
script_name 'msztime'
wait 2000

// int iTemp
int kReset,kExit // TODO: maybe use extra vars?
int sHour,sMin,sWeather
int cHour,cMin,cWeather

:start_of_script
if does_file_exist "cleo\msztime.ini"
then
kReset = 0x55
kReset = read_int_from_ini_file "cleo\msztime.ini" "mszcfg" "reset"
kExit = 0x49
kExit = read_int_from_ini_file "cleo\msztime.ini" "mszcfg" "exit"
sHour = -1
sHour = read_int_from_ini_file "cleo\msztime.ini" "time" "hour"
sMin = -1
sMin = read_int_from_ini_file "cleo\msztime.ini" "time" "min"
sWeather = -1
sWeather = read_int_from_ini_file "cleo\msztime.ini" "weather" "type"
else
goto @end_of_script
end

get_time_of_day cHour cMin
cWeather = get_current_weather

:time_loop
wait 750
if is_key_pressed kReset
then
while is_key_pressed kReset
wait 250
end
goto @start_of_script
end
if is_key_pressed kExit
then
while is_key_pressed kExit
wait 250
end
goto @take_a_break
end
if and
sHour > -1
sMin > -1
then
set_time_of_day sHour sMin
else
set_time_of_day cHour cMin
end
if sWeather > -1
then
force_weather_now sWeather
else
force_weather_now cWeather
end
goto @time_loop

:take_a_break
set_weather_to_appropriate_type_now
release_weather
while not is_key_pressed kExit
wait 1000
end
goto @start_of_script

//time_setup:

//return

//weather_setup:

//return 

:end_of_script
terminate_this_custom_script
