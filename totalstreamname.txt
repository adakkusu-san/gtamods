{$NOSOURCE}
{$USE CLEO+}
script_name 'namstrm'
wait 2000

int i,n,t,o
float f
string b
int m = get_label_pointer @the_names

:loop
wait 0

use_text_commands True
i = 0
t = 0
f = 40.0
while i < 78
n = get_number_of_instances_of_streamed_script i
if n > 0
then
gosub @text_setup
o = i * 8
b = read_struct_offset m o 8
if n > 9
then
display_text_formatted 490.0 f "%s = %d" b n
else
display_text_formatted 490.0 f "%s =  %d" b n
end
f += 10.0
t += n
end
i++
end
gosub @text_setup
if t > 9
then
display_text_formatted 490.0 20.0 "Total = %d" t
else
display_text_formatted 490.0 20.0 "Total =  %d" t
end
use_text_commands False
goto @loop

:text_setup
set_text_proportional False
set_text_right_justify True
return
